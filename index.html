Pengaturan Database

Masukkan URL endpoint dari deploy Code.gs agar data tersimpan ke Google Sheets Anda.

Script Code.gs (Google Sheets)

Cara Pemasangan:

  1. Buka spreadsheet baru di Google Sheets.
  2. Klik menu Ekstensi > Apps Script.
  3. Hapus semua kode lama, paste kode di bawah ini, lalu klik Simpan (💾).
  4. Pilih fungsi setupDatabase di bagian atas layar (sebelah tombol Jalankan), lalu klik Jalankan (Run) untuk membuat tabel otomatis. (Beri izin akses jika diminta).
  5. Klik tombol Terapkan (Deploy) > Deployment Baru.
  6. Pilih jenis Aplikasi Web. Akses: Siapa Saja (Anyone).
  7. Salin URL Web App yang muncul, lalu paste ke menu Set URL di aplikasi ini.
// Code.gs - Apps Script Backend for Attendance App
// Dideploy sebagai "Aplikasi Web" (Web App)

// Fungsi untuk membuat database otomatis di Google Sheets
function setupDatabase() {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  
  // 1. Setup Sheet Daftar_Siswa
  var rosterSheet = ss.getSheetByName("Daftar_Siswa") || ss.insertSheet("Daftar_Siswa");
  if (rosterSheet.getLastRow() === 0) {
    rosterSheet.appendRow(["ID", "NAMA SISWA", "JENIS KELAMIN"]);
    rosterSheet.getRange("A1:C1").setFontWeight("bold").setBackground("#e0f2fe");
    rosterSheet.setFrozenRows(1);
    rosterSheet.setColumnWidth(2, 250);
  }

  // 2. Setup Sheet Data_Absensi
  var attSheet = ss.getSheetByName("Data_Absensi") || ss.insertSheet("Data_Absensi");
  if (attSheet.getLastRow() === 0) {
    attSheet.appendRow(["KEY", "STATUS"]);
    attSheet.getRange("A1:B1").setFontWeight("bold").setBackground("#fef08a");
    attSheet.setFrozenRows(1);
    attSheet.setColumnWidth(1, 200);
  }

  // 3. Setup Sheet Catatan
  var noteSheet = ss.getSheetByName("Catatan") || ss.insertSheet("Catatan");
  if (noteSheet.getLastRow() === 0) {
    noteSheet.appendRow(["KEY", "CATATAN"]);
    noteSheet.getRange("A1:B1").setFontWeight("bold").setBackground("#fed7aa");
    noteSheet.setFrozenRows(1);
    noteSheet.setColumnWidth(1, 200);
    noteSheet.setColumnWidth(2, 300);
  }

  // 4. Hapus Sheet1 bawaan jika kosong
  var sheet1 = ss.getSheetByName("Sheet1");
  if (sheet1 && sheet1.getLastRow() === 0) {
    ss.deleteSheet(sheet1);
  }
  
  SpreadsheetApp.getUi().alert("Database Absensi berhasil dibuat secara otomatis!");
}

function doGet(e) {
  try {
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    
    // Read Roster
    var rosterSheet = ss.getSheetByName("Daftar_Siswa") || ss.insertSheet("Daftar_Siswa");
    var rosterData = [];
    if (rosterSheet.getLastRow() > 1) {
      var rVals = rosterSheet.getRange(2, 1, rosterSheet.getLastRow() - 1, 3).getValues();
      rVals.forEach(function(r) {
        if (r[0]) rosterData.push({ id: r[0], name: r[1], gender: r[2] });
      });
    }

    // Read Attendance
    var attSheet = ss.getSheetByName("Data_Absensi") || ss.insertSheet("Data_Absensi");
    var attendanceMap = {};
    if (attSheet.getLastRow() > 1) {
      var aVals = attSheet.getRange(2, 1, attSheet.getLastRow() - 1, 2).getValues();
      aVals.forEach(function(r) { if (r[0]) attendanceMap[r[0]] = r[1]; });
    }

    // Read Notes
    var noteSheet = ss.getSheetByName("Catatan") || ss.insertSheet("Catatan");
    var notesMap = {};
    if (noteSheet.getLastRow() > 1) {
      var nVals = noteSheet.getRange(2, 1, noteSheet.getLastRow() - 1, 2).getValues();
      nVals.forEach(function(r) { if (r[0]) notesMap[r[0]] = r[1]; });
    }

    return ContentService.createTextOutput(JSON.stringify({
      status: "success",
      students: rosterData,
      attendanceMap: attendanceMap,
      notesMap: notesMap
    })).setMimeType(ContentService.MimeType.JSON);

  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({ status: "error", message: err.toString() }))
      .setMimeType(ContentService.MimeType.JSON);
  }
}

function doPost(e) {
  try {
    var data = JSON.parse(e.postData.contents);
    var ss = SpreadsheetApp.getActiveSpreadsheet();

    // 1. Roster Update
    if (data.students) {
      var rosterSheet = ss.getSheetByName("Daftar_Siswa") || ss.insertSheet("Daftar_Siswa");
      rosterSheet.clear();
      rosterSheet.appendRow(["ID", "NAMA SISWA", "JENIS KELAMIN"]);
      rosterSheet.getRange("A1:C1").setFontWeight("bold").setBackground("#e0f2fe");
      if (data.students.length > 0) {
        var rRows = data.students.map(function(s) { return [s.id, s.name, s.gender]; });
        rosterSheet.getRange(2, 1, rRows.length, 3).setValues(rRows);
      }
    }

    // 2. Attendance Map Update
    if (data.attendanceMap) {
      var attSheet = ss.getSheetByName("Data_Absensi") || ss.insertSheet("Data_Absensi");
      attSheet.clear();
      attSheet.appendRow(["KEY", "STATUS"]);
      attSheet.getRange("A1:B1").setFontWeight("bold").setBackground("#fef08a");
      var attKeys = Object.keys(data.attendanceMap);
      if (attKeys.length > 0) {
        var aRows = attKeys.map(function(k) { return [k, data.attendanceMap[k]]; });
        attSheet.getRange(2, 1, aRows.length, 2).setValues(aRows);
      }
    }

    // 3. Notes Update
    if (data.notesMap) {
      var noteSheet = ss.getSheetByName("Catatan") || ss.insertSheet("Catatan");
      noteSheet.clear();
      noteSheet.appendRow(["KEY", "CATATAN"]);
      noteSheet.getRange("A1:B1").setFontWeight("bold").setBackground("#fed7aa");
      var noteKeys = Object.keys(data.notesMap);
      if (noteKeys.length > 0) {
        var nRows = noteKeys.map(function(k) { return [k, data.notesMap[k]]; });
        noteSheet.getRange(2, 1, nRows.length, 2).setValues(nRows);
      }
    }

    // 4. Monthly Summary Sheet (Optional but useful)
    if (data.month && data.studentsSummary) {
      var sheetName = "Rekap_" + data.month.replace(/\s+/g, '_');
      var summarySheet = ss.getSheetByName(sheetName) || ss.insertSheet(sheetName);
      summarySheet.clear();
      summarySheet.appendRow(["NO", "NAMA SISWA", "L/P", "HADIR (H)", "ALPA (A)", "SAKIT (S)", "IZIN (I)", "CATATAN"]);
      summarySheet.getRange("A1:H1").setFontWeight("bold").setBackground("#bbf7d0");
      
      if (data.studentsSummary.length > 0) {
        var sumRows = data.studentsSummary.map(function(s, idx) {
          return [idx + 1, s.name, s.gender, s.totals.H, s.totals.A, s.totals.S, s.totals.I, s.note];
        });
        summarySheet.getRange(2, 1, sumRows.length, 8).setValues(sumRows);
      }
    }

    return ContentService.createTextOutput(JSON.stringify({ status: "success" }))
      .setMimeType(ContentService.MimeType.JSON);

  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({ status: "error", message: err.toString() }))
      .setMimeType(ContentService.MimeType.JSON);
  }
}
