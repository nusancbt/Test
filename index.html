Integrasi Google Apps Script (GS)

Anda dapat menyalurkan aplikasi absensi ini langsung ke **Google Sheets** menggunakan Google Apps Script! Ikuti langkah mudah berikut:

*Masukkan URL Web App dari Google Apps Script agar tombol "Sync ke G-Sheet" berfungsi.

  1. Buka **Google Spreadsheet** baru di Google Drive Anda.
  2. Buka menu **Ekstensi > Apps Script**.
  3. Hapus kode bawaan, lalu salin dan tempel (copy-paste) kode backend dibawah ini pada file Code.gs.
  4. Klik **Terapkan (Deploy) > Terapkan sebagai web app (Deploy as web app)**.
  5. Pilih **Akses: Siapa saja (Anyone)**, lalu klik deploy. Salin URL Web App yang diberikan dan tempel pada kolom di atas.
// Code.gs - Google Apps Script Backend (API Mode)
function doPost(e) {
  try {
    // Parse data dari aplikasi HTML
    var data = JSON.parse(e.postData.contents);
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var sheetName = "Absensi_" + data.month.replace(/\s+/g, '_');
    
    // Cari atau buat sheet baru sesuai bulan
    var sheet = ss.getSheetByName(sheetName);
    if (!sheet) {
      sheet = ss.insertSheet(sheetName);
      sheet.appendRow(["NO", "NAMA SISWA", "JK", "BULAN", "HADIR", "SAKIT", "IZIN", "ALPA", "KETERANGAN"]);
      sheet.getRange("A1:I1").setFontWeight("bold").setBackground("#dbeafe");
    }
    
    // Hapus data lama agar tidak duplikat saat di-sync ulang
    if (sheet.getLastRow() > 1) {
      sheet.getRange(2, 1, sheet.getLastRow() - 1, 9).clearContent();
    }
    
    // Susun data baru
    var rows = [];
    data.students.forEach(function(s, idx) {
      rows.push([
        idx + 1, 
        s.name, 
        s.gender, 
        data.month,
        s.totals.H || 0,
        s.totals.S || 0,
        s.totals.I || 0,
        s.totals.A || 0,
        s.note || ""
      ]);
    });
    
    // Masukkan data ke sheet
    if (rows.length > 0) {
      sheet.getRange(2, 1, rows.length, 9).setValues(rows);
    }
    
    return ContentService.createTextOutput(JSON.stringify({ status: "success", message: "Data berhasil di-sync ke Google Sheets!" }))
      .setMimeType(ContentService.MimeType.JSON);
      
  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({ status: "error", message: err.toString() }))
      .setMimeType(ContentService.MimeType.JSON);
  }
}
