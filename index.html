Backend Script Google Sheets (Code.gs)

Petunjuk Pemasangan di Google Sheets:

  1. Buka Google Sheets Anda, pilih menu Ekstensi > Apps Script.
  2. Hapus semua kode lama, lalu Paste kode JavaScript di bawah ini.
  3. Klik tombol Deploy > Web app baru.
  4. Setel Execute as: Me dan Who has access: Anyone (Siapa Saja).
  5. Simpan Web App URL jika Anda ingin menghubungkan aplikasi.
// Code.gs - Web App Backend Cloud Database Google Sheets (Full Sync + Realtime Data)

function doGet(e) {
  try {
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    
    // Read Roster Sheet
    var rosterSheet = ss.getSheetByName("Daftar_Siswa") || ss.insertSheet("Daftar_Siswa");
    var rosterData = [];
    if (rosterSheet.getLastRow() > 1) {
      var rVals = rosterSheet.getRange(2, 1, rosterSheet.getLastRow() - 1, 3).getValues();
      rVals.forEach(function(r) {
        if (r[0]) rosterData.push({ id: r[0], name: r[1], gender: r[2] });
      });
    }

    // Read Attendance Sheet
    var attSheet = ss.getSheetByName("Data_Absensi") || ss.insertSheet("Data_Absensi");
    var attendanceMap = {};
    if (attSheet.getLastRow() > 1) {
      var aVals = attSheet.getRange(2, 1, attSheet.getLastRow() - 1, 2).getValues();
      aVals.forEach(function(r) {
        if (r[0]) attendanceMap[r[0]] = r[1];
      });
    }

    // Read Notes Sheet
    var noteSheet = ss.getSheetByName("Catatan") || ss.insertSheet("Catatan");
    var notesMap = {};
    if (noteSheet.getLastRow() > 1) {
      var nVals = noteSheet.getRange(2, 1, noteSheet.getLastRow() - 1, 2).getValues();
      nVals.forEach(function(r) {
        if (r[0]) notesMap[r[0]] = r[1];
      });
    }

    return ContentService.createTextOutput(JSON.stringify({
      status: "success",
      students: rosterData,
      attendanceMap: attendanceMap,
      notesMap: notesMap
    })).setMimeType(ContentService.MimeType.JSON);

  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({ status: "error", message: err.toString() }))
      .setMimeType(ContentService.MimeType.JSON);
  }
}

function doPost(e) {
  try {
    var data = JSON.parse(e.postData.contents);
    var ss = SpreadsheetApp.getActiveSpreadsheet();

    // 1. Update Roster Sheet
    if (data.students) {
      var rosterSheet = ss.getSheetByName("Daftar_Siswa") || ss.insertSheet("Daftar_Siswa");
      rosterSheet.clear();
      rosterSheet.appendRow(["ID", "NAMA SISWA", "JENIS KELAMIN"]);
      rosterSheet.getRange("A1:C1").setFontWeight("bold").setBackground("#dbeafe");
      if (data.students.length > 0) {
        var rRows = data.students.map(function(s) { return [s.id, s.name, s.gender]; });
        rosterSheet.getRange(2, 1, rRows.length, 3).setValues(rRows);
      }
    }

    // 2. Update Attendance Map Sheet
    if (data.attendanceMap) {
      var attSheet = ss.getSheetByName("Data_Absensi") || ss.insertSheet("Data_Absensi");
      attSheet.clear();
      attSheet.appendRow(["KEY", "STATUS"]);
      attSheet.getRange("A1:B1").setFontWeight("bold").setBackground("#dbeafe");
      var attKeys = Object.keys(data.attendanceMap);
      if (attKeys.length > 0) {
        var aRows = attKeys.map(function(k) { return [k, data.attendanceMap[k]]; });
        attSheet.getRange(2, 1, aRows.length, 2).setValues(aRows);
      }
    }

    // 3. Update Notes Map Sheet
    if (data.notesMap) {
      var noteSheet = ss.getSheetByName("Catatan") || ss.insertSheet("Catatan");
      noteSheet.clear();
      noteSheet.appendRow(["KEY", "CATATAN"]);
      noteSheet.getRange("A1:B1").setFontWeight("bold").setBackground("#dbeafe");
      var noteKeys = Object.keys(data.notesMap);
      if (noteKeys.length > 0) {
        var nRows = noteKeys.map(function(k) { return [k, data.notesMap[k]]; });
        noteSheet.getRange(2, 1, nRows.length, 2).setValues(nRows);
      }
    }

    // 4. Generate Monthly Summary Sheet (Optional Export View)
    if (data.month && data.studentsSummary) {
      var sheetName = "Rekap_" + data.month.replace(/\s+/g, '_');
      var summarySheet = ss.getSheetByName(sheetName) || ss.insertSheet(sheetName);
      summarySheet.clear();
      summarySheet.appendRow(["NO", "NAMA SISWA", "JK", "HADIR", "ALPA", "SAKIT", "IZIN", "KETERANGAN"]);
      summarySheet.getRange("A1:H1").setFontWeight("bold").setBackground("#bbf7d0");
      
      var sumRows = data.studentsSummary.map(function(s, idx) {
        return [idx + 1, s.name, s.gender, s.totals.H, s.totals.A, s.totals.S, s.totals.I, s.note];
      });
      if (sumRows.length > 0) {
        summarySheet.getRange(2, 1, sumRows.length, 8).setValues(sumRows);
      }
    }

    return ContentService.createTextOutput(JSON.stringify({ status: "success", message: "Data tersimpan ke Google Sheets!" }))
      .setMimeType(ContentService.MimeType.JSON);

  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({ status: "error", message: err.toString() }))
      .setMimeType(ContentService.MimeType.JSON);
  }
}
