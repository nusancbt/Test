Integrasi Google Apps Script (GS)

Anda juga dapat menyalurkan aplikasi absensi ini langsung ke **Google Sheets** menggunakan Google Apps Script! 

// Code.gs - Google Apps Script Backend (API Mode)
function doPost(e) {
  try {
    var data = JSON.parse(e.postData.contents);
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var sheetName = "Absensi_" + data.month.replace(/\s+/g, '_');
    
    var sheet = ss.getSheetByName(sheetName);
    if (!sheet) {
      sheet = ss.insertSheet(sheetName);
      sheet.appendRow(["NO", "NAMA SISWA", "JK", "BULAN", "HADIR", "SAKIT", "IZIN", "ALPA", "KETERANGAN"]);
      sheet.getRange("A1:I1").setFontWeight("bold").setBackground("#dbeafe");
    }
    
    if (sheet.getLastRow() > 1) {
      sheet.getRange(2, 1, sheet.getLastRow() - 1, 9).clearContent();
    }
    
    var rows = [];
    data.students.forEach(function(s, idx) {
      rows.push([
        idx + 1, 
        s.name, 
        s.gender, 
        data.month,
        s.totals.H || 0,
        s.totals.S || 0,
        s.totals.I || 0,
        s.totals.A || 0,
        s.note || ""
      ]);
    });
    
    if (rows.length > 0) {
      sheet.getRange(2, 1, rows.length, 9).setValues(rows);
    }
    
    return ContentService.createTextOutput(JSON.stringify({ status: "success", message: "Data berhasil di-sync ke Google Sheets!" }))
      .setMimeType(ContentService.MimeType.JSON);
      
  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({ status: "error", message: err.toString() }))
      .setMimeType(ContentService.MimeType.JSON);
  }
}
